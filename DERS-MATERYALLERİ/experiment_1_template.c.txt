int read_int(const char *prompt, int min, int max) {
    int value;

    printf("%s", prompt);
    while (scanf("%d", &value) != 1 || value < min || value > max) {
        while (getchar() != '\n');
        printf("Invalid input. Enter a value from %d to %d: ", min, max);
    }
    return value;
}

double read_double(const char *prompt, double min, double max) {
    double value;

    printf("%s", prompt);
    while (scanf("%lf", &value) != 1 || value < min || value > max) {
        while (getchar() != '\n');
        printf("Invalid input. Enter a value from %.2f to %.2f: ", min, max);
    }
    return value;
}
